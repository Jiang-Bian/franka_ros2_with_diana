cmake_minimum_required(VERSION 3.16)

project(diana_test)

find_package(ament_cmake REQUIRED)

find_package(franka REQUIRED)

add_executable(
    diana_libfranka_compatibility_test
    diana_libfranka_compatibility_test.cpp
)

        lib/${PROJECT_NAME}
